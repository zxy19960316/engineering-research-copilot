### Task 2: Add the Raw-SHA EOL Regression Gate Red-First

**Files:**
- Create: `tests/test_m3_raw_sha_eol_policy.py`

**Interfaces:**
- Consumes: the frozen r5.1-f02 manifest, its input-binding JSON, Git attributes, current worktree bytes, and HEAD blob bytes.

- [x] **Step 1: Create the focused and root-wide regression tests**

Use this exact binding scope, matching the historical auditor's worktree reads:

```python
from __future__ import annotations

import json
import subprocess
import unittest
from pathlib import Path


REPO_ROOT = Path(__file__).resolve().parents[1]
MANIFEST = REPO_ROOT / "evals/m3/forward-inputs-r5.1-f02/manifest.json"


class M3RawShaEolPolicyTests(unittest.TestCase):
    def _attributes(self, relative: str) -> dict[str, str]:
        raw = subprocess.run(
            ["git", "check-attr", "-z", "text", "eol", "--", relative],
            cwd=REPO_ROOT,
            capture_output=True,
            check=True,
        ).stdout
        fields = raw.decode("utf-8").rstrip("\0").split("\0")
        return {fields[index + 1]: fields[index + 2] for index in range(0, len(fields), 3)}

    def _bindings(self) -> list[dict[str, str]]:
        manifest = json.loads(MANIFEST.read_text(encoding="utf-8"))
        bindings = [manifest[key] for key in ("input_binding", "prompt", "contract", "supersession_policy")]
        input_binding = json.loads((REPO_ROOT / manifest["input_binding"]["path"]).read_text(encoding="utf-8"))
        bindings.extend(input_binding[key] for key in ("source_input", "m2_validation", "eligibility"))
        return bindings

    def test_all_r5_1_preparation_worktree_raw_sha_bindings_use_lf(self) -> None:
        bindings = self._bindings()
        paths = [binding["path"] for binding in bindings]
        self.assertEqual(len(paths), 7)
        self.assertEqual(len(set(paths)), 7)
        for relative in paths:
            with self.subTest(path=relative):
                self.assertEqual(self._attributes(relative), {"text": "set", "eol": "lf"})
                worktree = (REPO_ROOT / relative).read_bytes()
                blob = subprocess.run(
                    ["git", "show", f"HEAD:{relative}"],
                    cwd=REPO_ROOT,
                    capture_output=True,
                    check=True,
                ).stdout
                self.assertEqual(worktree, blob)
```

Also enumerate `git ls-files -z -- evals/m3/results/diagnostics-r5.1` and apply the same `text=set`, `eol=lf`, and HEAD-blob byte-equality assertions to every returned tracked path. Require at least the two existing diagnostics files so an accidentally empty enumeration cannot pass.
